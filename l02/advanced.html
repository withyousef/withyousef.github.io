تمارين الورق — الأنواع والتحويل

التمارين دى تُحلّ بالقلم على الورق، من غير تابلت ومن غير تشغيل. الهدف إن الطالب يقرا السطر ويقول نوعه قبل ناتجه — دى العادة اللى بتمنع نص أخطاء الامتحان. الحلول مخفية؛ افتحها بعد ما تجرّب.

إعداد: يوسف زكريا التهامى · تأسيس البرمجة — بكالوريا ٢٠٢٧

الاسمالتاريخ
ورقة الطلاب بتطلع بمساحة إجابة وبدون حلول.
المستوى المتقدم — للى خلّص بدرى
٠٥

تتبع بالتحويل

٥ دقائق
let a = "7";
let b = Number(a);
let c = String(b);

console.log(a + b);
console.log(b + b);
console.log(c + b);
console.log(typeof c);

اكتب نوع كل متغير من الثلاثة، وبعدين ناتج كل سطر طباعة.

الحل

الأنواع: a نص string · b رقم number · c نص string تانى.

الناتج: 77 ثم 14 ثم 77 ثم string

  1. a + b — طرف نص، يبقى لزق: "7" و7 يبقوا 77.
  2. b + b — الطرفان رقمان، يبقى جمع: ٧ + ٧ = 14.
  3. c + b — c رجع نصاً تانى، فرجعنا للزق: 77.

الفخ هنا: الطالب فاكر إن Number() غيّرت a نفسها. ما غيّرتش حاجة — هى رجّعت قيمة جديدة اتخزّنت فى b، وa لسه نصاً زى ما هو.

٠٦

فين الغلط؟ ثلاثة أغلاط، حدّدهم وصحّحهم

٥ دقائق
let mark = "80";
let bonus = 5;
let total = mark + bonus;
console.log("Total = " Total);

المطلوب إن الكود يطبع Total = 85.

الحل
  1. السطر ٣: mark نص، فالعلامة + بتلزّق والناتج 805. الصح: let total = Number(mark) + bonus;
  2. السطر ٤: ناقص علامة + بين النص والمتغير. الصح: "Total = " + total
  3. السطر ٤: Total بحرف كبير — ده اسم تانى خالص ومش موجود. الصح total بحرف صغير.
let mark = "80";
let bonus = 5;
let total = Number(mark) + bonus;
console.log("Total = " + total);
٠٧

اكتب برنامجاً كاملاً

٦ دقائق

طالب جاب ثلاث درجات، وجاتك من خانة إدخال فوصلت كلها نصوصاً: "20" و"18" و"25". اكتب كوداً يحطّ كل درجة فى متغير، ويحسب المجموع صح فى متغير رابع، ويطبع جملة على شكل Total = 63. وفى الآخر اطبع نوع المجموع علشان تتأكد.

الحل 
let mark1 = "20";
let mark2 = "18";
let mark3 = "25";
let total = Number(mark1) + Number(mark2) + Number(mark3);
console.log("Total = " + total);
console.log(typeof total);   // number

حل تانٍ مقبول تماماً: يحوّل كل واحدة فى سطر مستقل قبل الجمع — let n1 = Number(mark1); وهكذا. ده أطول بس أوضح، وامدحه لو طالب عمله.

الغلط المتوقع: Number(mark1 + mark2 + mark3) — دى بتلزّق الأول فتطلع "201825" وبعدين بتحوّلها لرقم واحد كبير. اكتبها على السبورة وخلّيهم يشوفوا الناتج، أقوى من أى شرح.

المستوى الأساسى — مطلوب من الكل

التمارين دى مكانها الطبيعى فى بلوك الشرح نفسه، بس محطوطة هنا علشان كل ورق المحاضرة يبقى فى صفحة واحدة.

٠١

إيه النوع؟

دقيقتان
console.log(typeof 12);
console.log(typeof "12");
console.log(typeof true);
console.log(typeof "true");
الحل

number ثم string ثم boolean ثم string

السطر الأخير هو الفخ: "true" بين تنصيص، يبقى كلمة مكتوبة مش قيمة منطقية. علامتا التنصيص بتغلبا أى حاجة.

٠٢

جمع ولا لزق؟

٣ دقائق
console.log("3" + 4);
console.log(3 + 4);
console.log("3" + "4");

اكتب ناتج كل سطر، وجنبه كلمة «جمع» أو «لزق».

الحل

34 لزق · 7 جمع · 34 لزق

السطر الأول والتالت ناتجهما واحد، والسبب واحد: وجود نص. طرف واحد نص كفاية.

٠٣

صحّح السطر

دقيقتان

السطر ده بيطبع 105، والمطلوب إنه يطبع 15. عدّله.

console.log("10" + 5);
الحل 
console.log(Number("10") + 5);

مقبول كمان: شيل علامتى التنصيص — console.log(10 + 5);. الحلان صح، بس الأول هو اللى بينفع لما الرقم جاى لك من برّه ومش متحكم فيه.

٠٤

املأ جدول الناتج والنوع

٣ دقائق
السطرالناتجنوع الناتج
Number("8")
String(8)
Number("eight")
الحل

Number("8") بترجّع 8 ونوعه number.

String(8) بترجّع "8" ونوعه string — وبتتطبع على الشاشة 8 من غير تنصيص، وده اللى بيلخبط.

Number("eight") بترجّع NaN لأن الكلمة مش رقم. وسيبها كده فى الجدول؛ نوعها موضوع تانى مش مطلوب دلوقتى.
