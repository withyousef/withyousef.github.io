القاعدة الوحيدة

جاوب الأول، وبعدين اكشف. لو كشفت قبل ما تفكّر، التمرين بيبقى ملوش أى لازمة — إنت بتخدع نفسك مش المدرس.

و«الصندوق» هنا هو نفسه «الدرج» بتاع المحاضرة، وهو نفسه اللى الكتب بتسمّيه متغير.

القسم الأول · ٨ أسئلة

اختار

٠١

سلة مشتريات — تابع total سطراً سطراً

اختيار من متعدد

اقرا البرنامج كله الأول، وبعدين قرّر. خد بالك من نوع كل صندوق.

let item = "Book";
let price = "45";      // جاى من خانة كتب فيها المستخدم
let qty = 3;
let total = price * qty;
console.log(total);
console.log(typeof total);
total = total + "0";
console.log(total);
console.log(typeof total);
console.log(item + " = " + total);

الإجابة الصح هى أ

الناتج135 number 1350 string Book = 1350

السطر الرابع: "45" * 3. علامة الضرب مالهاش غير معنى واحد، فبتحوّل النص لرقم لوحدها والناتج 135 رقم. وعشان كده typeof قالت number.

والسطر السابع هو نقطة الانقلاب: 135 + "0". علامة الزائد ليها معنيان، ولما شافت نصاً لزقت — فطلع النص "1350" مش الرقم ١٣٥.

خد بالك إن القيمة زادت عشر مرات من غير أى عملية حسابية. ده مش خطأ فى الحساب — ده لزق.

والدرس: * و- و/ بتحوّل، و+ بس هى اللى بتلزّق. لو ما عرفتش النوع، ما تعرفش الناتج.

الشرح خطوة بخطوة
٠١

الضرب بيحوّل لوحده

let total = price * qty;

وده اللى بيخلّى الغلط ده صعب: البرنامج اشتغل صح، والنوع طلع number فعلاً.

٠٢

والزائد بيلزّق

total = total + "0";

صفر واحد بين تنصيص خلّى الرقم يتحوّل لنص ويكبر عشر مرات.

٠٣

والنوع بيفضح الفرق

console.log(typeof total);   // number قبل، وstring بعد

لو ما كتبتش typeof، كنت هتفتكر إن البرنامج شغّال — وهو شغّال بنوع غلط.

٠٢

عدّاد بيتراكم — وسطر واحد بيقلبه

اختيار من متعدد

أربع خطوات تراكم، واحدة منهم بس فيها تنصيص. إيه اللى هيتطبع؟

let score = 0;
score = score + 10;
score = score + 10;
console.log(score);
score = score + "10";
console.log(score);
score = score + 10;
console.log(score);
console.log(typeof score);

الإجابة الصح هى د

الناتج20 2010 201010 string

أول خطوتين تراكم عادى: صفر زائد عشرة زائد عشرة يساوى 20 رقم.

والسطر الخامس فيه "10" بتنصيص، فالزائد لزق: 20 بقت "2010" — نص.

ومن دى ورايح، الصندوق نص. فالسطر السابع "2010" + 10 لزق برضه وطلع "201010".

ودى الحاجة اللى لازم تفهمها: التلوّث بيفضل. سطر واحد بتنصيص بيقلب نوع الصندوق، وكل اللى بعده بياخد النوع الجديد.

الشرح خطوة بخطوة
٠١

التراكم عادى لحد هنا

score = score + 10;   // صفر ثم عشرة ثم عشرين

كل حاجة أرقام، فالزائد بيجمع.

٠٢

وسطر واحد بيقلب النوع

score = score + "10";

لاحظ إن 2010 مش عشرين زائد عشرة — دى «٢٠» و«١٠» ملزوقين ورا بعض.

٠٣

والتلوّث بيفضل

score = score + 10;   // دلوقتى نص + رقم = لزق

نفس السطر اللى كان بيجمع فى الأول بقى بيلزّق — لأن الصندوق اتغيّر نوعه، مش لأن السطر اتغيّر.

٠٣

أربع تحويلات — واحدة منهم بتفشل

اختيار من متعدد

كل سطر بيحوّل حاجة. إيه اللى هيتطبع؟

let a = Number("25");
let b = Number("25kg");
let c = String(25);
let d = Number("");
console.log(a);
console.log(b);
console.log(c + 1);
console.log(d);
console.log(typeof b);

الإجابة الصح هى و

الناتج25 NaN 251 0 number

Number("25") نجحت لأن النص كله أرقام.

Number("25kg") فشلت وطلّعت NaN. وده أهم سطر فى التمرين: Number() مش بتاخد الجزء الرقمى وتسيب الباقى — يا كله يا لأ.

String(25) عملت النص "25"، فـc + 1 لزق وطلع 251 مش 26.

وNumber("") النص الفاضى بتطلّع صفراً مش NaN — ودى غريبة بس لازم تفتكرها.

وآخر سطر: typeof NaN بيطلّع number. آه، النوع رقم. NaN معناها «رقم بس مش معروف كام»، مش «مش رقم».

الشرح خطوة بخطوة
٠١

التحويل الناجح

let a = Number("25");

الشرط الوحيد: النص يكون كله أرقام (والمسافات اللى حواليه مسموحة).

٠٢

والتحويل الفاشل

let b = Number("25kg");

ما بتاخدش ٢٥ وتسيب kg. لو فيه حرف واحد مش رقم، التحويل كله بيفشل.

٠٣

وString بتعمل العكس

let c = String(25);
console.log(c + 1);   // 251 مش 26

String() بتحوّل الرقم لنص — وبعدها الزائد بيلزّق مش بيجمع.

٠٤

خمس مقارنات — الفرق بين علامتين وثلاثة

اختيار من متعدد

قارن القيمة، وقارن النوع. إيه اللى هيتطبع؟

let x = 5;
let y = "5";
console.log(x == y);
console.log(x === y);
console.log(typeof x === typeof y);
console.log(x === Number(y));
let z = x + 0;
console.log(z === x);

الإجابة الصح هى أ

الناتجtrue false false true true

x == y بتقارن القيمة بس، فبتحوّل قبل ما تقارن — والاتنين خمسة فطلعت true.

x === y بتقارن القيمة والنوع. القيمة واحدة بس النوع مختلف (رقم ونص)، فطلعت false.

والسطر التالت بيقارن الأنواع نفسها: "number" === "string" يعنى false.

والسطر الرابع: بعد Number(y) بقى رقماً، فالمقارنة الصارمة نجحت.

والأخير: 5 + 0 يساوى 5 رقماً، فـz === x صحيحة.

القاعدة العملية: استعمل === دايماً. == بتخبّى فروق الأنواع، والفروق دى هى اللى بتطلّع الأخطاء الصعبة.

الشرح خطوة بخطوة
٠١

العلامتان بتحوّلا قبل المقارنة

console.log(x == y);    // true
console.log(x === y);   // false

دى الحتة اللى بتلخبط: نفس القيمتين بيدّوا ناتجين مختلفين حسب العلامة.

٠٢

وtypeof بترجّع نصاً

console.log(typeof x === typeof y);   // "number" === "string"

typeof ما بتقارنش — هى بترجّع كلمة، وإنت اللى بتقارن الكلمتين.

٠٥

ثلاث درجات جايّة نصاً — والمجموع غلط

اختيار من متعدد

برنامج بيحسب مجموع تلات درجات. إيه اللى هيتطبع؟

let m1 = "20";
let m2 = "15";
let m3 = 10;
let sum = m1 + m2 + m3;
console.log(sum);
console.log(typeof sum);
let real = Number(m1) + Number(m2) + m3;
console.log(real);
console.log(sum == real);
console.log(sum === real);

الإجابة الصح هى هـ

الناتج201510 string 45 false false

"20" + "15" نصان، فالزائد لزق: "2015". وبعدين "2015" + 10 لزق برضه: "201510".

ودى مصيبة صامتة: البرنامج ما وقعش بخطأ، والناتج شكله رقم — بس هو نص، والمجموع غلط تماماً.

والسطر السابع صحّح الوضع بـNumber() على الاتنين، فطلع 45 صح.

وآخر سطرين بيقارنوا "201510" مع 45 — لا القيمة واحدة ولا النوع، فالاتنين false.

الدرس: أى حاجة جاية من المستخدم بتيجى نصاً. حوّلها قبل ما تجمع، مش بعد.

الشرح خطوة بخطوة
٠١

نصان بيتلزقوا

let sum = m1 + m2 + m3;

الناتج 201510 — رقم شكله كبير، بس هو مجرد تلات حتت ملزوقين.

٠٢

والتحويل قبل الجمع بيصلّح

let real = Number(m1) + Number(m2) + m3;

لاحظ إن m3 ما احتاجش تحويل — هو رقم أصلاً.

٠٣

والمقارنة بتفضح الفرق

console.log(sum == real);    // false
console.log(sum === real);   // false

هنا حتى == طلعت false، لأن الفرق مش فى النوع بس — الحساب نفسه غلط.

٠٦

NaN بيعدّى — تابعه لآخر سطر

اختيار من متعدد

صندوق واحد فشل تحويله، وبعدين اتحسب عليه مرتين.

let raw = "abc";
let n = Number(raw);
console.log(n);
n = n + 10;
console.log(n);
n = n * 2;
console.log(n);
console.log(typeof n);
console.log(n === n);

الإجابة الصح هى د

الناتجNaN NaN NaN number false

Number("abc") فشلت فطلّعت NaN.

ومن هنا، أى عملية حسابية على NaN بتطلّع NaN. زوّد عشرة: NaN. اضرب فى اتنين: NaN. البرنامج بيكمّل شغل من غير ما يقع، والنتيجة بايظة من أول سطر.

وtypeof NaN بيطلّع number. الاسم مضلّل: NaN مش «مش رقم» بالمعنى ده — هى قيمة رقمية معناها «الحساب ده مالوش نتيجة».

وآخر سطر أغرب حاجة فى اللغة: NaN === NaN بتطلّع false. NaN مش بتساوى نفسها.

الدرس العملى: اتأكد من التحويل ساعة ما يحصل، مش بعد عشر سطور.

الشرح خطوة بخطوة
٠١

التحويل بيفشل

let n = Number("abc");

ما وقعش بخطأ. ده أخطر من الخطأ — الخطأ بيوقّفك، وده بيسيبك ماشى غلط.

٠٢

وبيعدّى لكل اللى بعده

n = n + 10;   // NaN
n = n * 2;    // NaN

مفيش عملية بترجّعك من NaN. لازم تصلّح المصدر.

٠٣

وهى مش بتساوى نفسها

console.log(n === n);   // false

دى القيمة الوحيدة فى اللغة اللى بتتصرف كده.

٠٧

صندوق واحد بيغيّر نوعه أربع مرات

اختيار من متعدد

تابع v سطراً سطراً: القيمة والنوع.

let v = 5;
console.log(typeof v);
v = "5";
console.log(typeof v);
v = Number(v);
console.log(typeof v);
v = String(v);
console.log(typeof v);
v = v + 5;
console.log(v);
console.log(typeof v);

الإجابة الصح هى و

الناتجnumber string number string 55 string

الصندوق هنا غيّر نوعه أربع مرات، وما وقعش بخطأ ولا مرة. النوع مش بيتثبّت مع let — هو صفة اللى جوه الصندوق دلوقتى، مش صفة الصندوق نفسه.

ولاحظ إن القيمة المطبوعة فضلت 5 فى كل الخطوات، والنوع بس هو اللى بيتغيّر. الشاشة ما بتقولش لك النوع.

وآخر سطرين: بعد String() بقى نصاً، فـ"5" + 5 لزق وطلع 55.

لو كان فضل رقماً كان طلع 10. نفس السطر بالظبط، وناتجان مختلفان — والفرق كله فى النوع.

الشرح خطوة بخطوة
٠١

النوع صفة اللى جوه، مش صفة الصندوق

let v = 5;
v = "5";

ولا خطأ ولا تحذير. اللغة بتسمح، وعليك إنت إنك تعرف.

٠٢

التحويل رايح وجاى

v = Number(v);   // نص ← رقم
v = String(v);   // رقم ← نص

الدالتان عكس بعض، والقيمة رجعت لنفس مكانها — بس النوع هو اللى بيحدّد اللى جاى.

٠٣

وآخر سطر بيدفع الحساب

v = v + 5;   // "5" + 5 = "55"

لو السطر اللى قبله ما كانش موجود، كان الناتج 10.

٠٨

التمرين الأصعب — رصيد وخصم ومقارنة

اختيار من متعدد

اقرا البرنامج كله بتأنٍّ. كل سطر بيبنى على اللى قبله.

let balance = "500";
let fee = 50;
balance = balance - fee;
console.log(balance);
console.log(typeof balance);
balance = balance + fee;
console.log(balance);
balance = balance + "0";
console.log(balance);
console.log(typeof balance);
console.log(balance == 5000);
console.log(balance === 5000);

الإجابة الصح هى هـ

الناتج450 number 500 5000 string true false

السطر التالت: "500" - 50. علامة الناقص مالهاش غير الطرح، فحوّلت النص لرقم لوحدها — والناتج 450 رقم. الصندوق اتشفى من غير ما نكتب Number().

والسطر السادس: 450 + 50. دلوقتى الاتنين أرقام، فالزائد جمع عادى وطلع 500.

والسطر التامن: 500 + "0". الزائد شاف نصاً فلزق، والناتج "5000" — نص.

وآخر سطرين: "5000" == 5000 بتحوّل قبل ما تقارن فطلعت true، و=== بتقارن النوع كمان فطلعت false.

خلاصة الورقة كلها: الناقص والضرب والقسمة بتحوّل، والزائد بس هو اللى بيلزّق. ونفس القيمة ممكن تبقى true بعلامتين وfalse بثلاثة.

الشرح خطوة بخطوة
٠١

الناقص بيحوّل لوحده

balance = balance - fee;   // "500" - 50 = 450 رقم

ودى مفاجأة لناس كتير: الطرح «صلّح» النوع من غير ما حد يطلب منه.

٠٢

والزائد بيرجّع اللخبطة

balance = balance + "0";   // 500 + "0" = "5000"

صفر واحد بين تنصيص ضرب الرصيد فى عشرة. ودى الحاجة اللى بتوقّع البرامج الحقيقية.

٠٣

والمقارنتان بيدّوا ناتجين

console.log(balance == 5000);    // true
console.log(balance === 5000);   // false

لو كنت كتبت == فى برنامج حقيقى، كان هيعدّى وإنت فاكر إن كل حاجة تمام.

٠٩

المشروع الأول — كارت منتج بيتحقق من مدخلاته

كتابة

اكتب برنامجاً كاملاً بالمواصفات دى: ١ — صندوق name فيه "Mouse"، وصندوق price فيه النص "120"، وصندوق qty فيه الرقم ٢. ٢ — اطبع نوع price عشان تثبت إنه نص. ٣ — صندوق total فيه ناتج الضرب كرقم، ومكتوب فيه Number() صراحةً. ٤ — اطبع total ونوعه. ٥ — صندوق vat فيه ناتج ضرب total فى ٠٫١٤. ٦ — صندوق grand فيه total زائد vat. ٧ — وفى الآخر اطبع سطراً بالشكل ده بالظبط: Mouse x2 = 273.6

اتحفظت 
let name = "Mouse";
let price = "120";
let qty = 2;
console.log(typeof price);
let total = Number(price) * qty;
console.log(total);
console.log(typeof total);
let vat = total * 0.14;
let grand = total + vat;
console.log(name + " x" + qty + " = " + grand);
الناتجstring 240 number Mouse x2 = 273.6

السطر الخامس هو المفتاح: Number(price) * qty. علامة الضرب كانت هتحوّل لوحدها، بس كتابة Number() بتقول للى يقرا كودك إن إنت عارف إن ده نص وحوّلته عن قصد.

والفرق ده بيبان لما تسيب الضرب وتجيب الزائد: لو كتبت price + qty من غير تحويل، هتلاقى "1202" بدل ٢٤٢.

وفى السطرين اللى بعدهم كل حاجة أرقام، فالحساب طبيعى: 240 × 0.14 = 33.6، والمجموع 273.6.

وفى السطر الأخير اللزق بيوصّل النص بالرقم تلقائياً. عدّ المسافات: " x" فيها مسافة قبل، و" = " فيها مسافة قبل وبعد.

وخد بالك إن total فى السطر الأخير رقم، فاللزق هنا مقصود ومظبوط — مش غلط زى اللى فى التمارين اللى فاتت. اللزق مش عيب فى ذاته، العيب إنه يحصل من غير ما تقصده.

الشرح خطوة بخطوة
٠١

السعر بييجى نصاً — وده الوضع الطبيعى

let price = "120";
console.log(typeof price);   // string

دى مش حالة استثنائية — دى الحالة الافتراضية فى أى برنامج حقيقى بياخد مدخلات.

٠٢

التحويل الصريح قبل الحساب

let total = Number(price) * qty;

اكتب Number() حتى لو العلامة هتحوّل لوحدها — عشان اللى بيقرا كودك يفهم نيّتك.

٠٣

واللزق فى الآخر مقصود

console.log(name + " x" + qty + " = " + grand);

الفرق بين اللزق الصح واللزق الغلط هو إنك قصدته ولا لأ.

١٠

المشروع التانى — صلّح البرنامج المكسور

كتابة

البرنامج ده المفروض يحسب مجموع درجتين ويقول لو الطالب عدّى المئة. وهو شغّال من غير ما يقع بخطأ، بس ناتجه غلط: let a = "60"; let b = "45"; let total = a + b; console.log(total); console.log(total == 105);

المطلوب منك تلات حاجات:

١ — اكتب اللى بيطبعه دلوقتى بالظبط، سطرين. ٢ — اكتب البرنامج مصلّحاً بحيث يطبع 105 وtrue، وبشرط إن a وb يفضلوا نصوصاً زى ما هما (لأنهم جايّين من المستخدم). ٣ — قول فى سطر واحد ليه استعمال === بدل == كان هيكشف الغلط من الأول.

اتحفظت 
let a = "60";
let b = "45";
let total = Number(a) + Number(b);
console.log(total);
console.log(total === 105);
الناتج الحالى6045 false

الناتج الحالى: 6045 وبعدين false. النصان اتلزقوا فطلع "6045"، والمقارنة بـ105 طلعت false لأن حتى القيمة مختلفة.

التصحيح: Number(a) + Number(b). لازم التحويل يحصل قبل الجمع، مش بعده — لو حوّلت النتيجة Number("6045") هتطلع ٦٠٤٥ مش ١٠٥.

وليه ===: لو كان الكود الأصلى بيقارن بـ== وكانت القيمة صح بالصدفة، == كانت هتقول true وتخبّى إن النوع نص. و=== بتقارن النوع كمان، فكانت هتقول false وتخلّيك تدوّر على السبب.

وفى المثال ده الاتنين طلعوا false لأن الغلط كبير أوى (٦٠٤٥ مش ١٠٥). بس فى الحالات اللى الغلط فيها بسيط، == هى اللى بتخلّى الخطأ يعدّى.

القاعدة: حوّل عند المصدر، وقارن بـ===. الاتنين مع بعض بيمنعوا أغلب أخطاء الأنواع.

الناتج بعد التعديل105 true
الشرح خطوة بخطوة
٠١

الغلط: نصان بيتلزقوا

let total = a + b;   // "60" + "45" = "6045"

مفيش رسالة خطأ، ومفيش تحذير. الرقم شكله رقم، والبرنامج ماشى.

٠٢

التصحيح: حوّل قبل ما تجمع

let total = Number(a) + Number(b);   // 60 + 45 = 105

ترتيب الخطوات مهم: التحويل قبل الجمع. لو جمعت الأول خلاص راح.

٠٣

والمقارنة الصارمة بتكشف بدرى

console.log(total === 105);

الرسم ده بيوضّح الحالة اللى الغلط فيها بسيط: القيمة صح والنوع غلط. == بتعدّيها، و=== بتمسكها.
