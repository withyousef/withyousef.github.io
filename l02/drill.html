القاعدة الوحيدة

جاوب الأول، وبعدين اكشف. لو كشفت قبل ما تفكّر، التمرين بيبقى ملوش أى لازمة — إنت بتخدع نفسك مش المدرس.

و«الصندوق» هنا هو نفسه «الدرج» بتاع المحاضرة، وهو نفسه اللى الكتب بتسمّيه متغير.

القسم الأول · ٢٨ سؤالاً

إيه اللى هيتطبع؟

اختيار من متعدد 
let bags = 4;
bags = bags * 3;
bags = bags - 2;
console.log(bags);

الإجابة الصح هى أ

الناتج10

اقرا، احسب، خزّن — تلات مرات. ٤ ثم ١٢ ثم ١٠.

الفخ الوحيد إنك تحسب 4 - 2 الأول. الكمبيوتر بينفّذ سطراً سطراً بالترتيب.

الشرح خطوة بخطوة
٠١

السطر التانى: ضرب

bags = bags * 3;

اليمين بيتحسب الأول، وبعدين الناتج بيرجع للصندوق.

٠٢

السطر التالت: طرح من الجديد

bags = bags - 2;

الصندوق فيه ١٢ دلوقتى مش ٤ — فالطرح بيحصل من ١٢.

٠٢

إيه اللى هيتطبع؟

اختيار من متعدد 
let day = "Sat";
let copy = day;
day = "Sun";
console.log(copy);

الإجابة الصح هى د

الناتجSat

السطر التانى ناخد نسخة من اللى جوه day فى اللحظة دى، وحطّها فى copy.

وبعد كده تغيير day مالوش أى علاقة بـcopy — الصندوقين مستقلين عن بعض.

الشرح خطوة بخطوة
٠١

النسخ بيحصل مرة واحدة

let copy = day;

الصورة الصح: نقلت ورقة من صندوق لصندوق، مش وصّلت الصندوقين بماسورة.

٠٢

التغيير بيلمس صندوقاً واحداً

day = "Sun";

لو كان فيه ربط بينهم، كان copy بقى Sun كمان — بس مفيش ربط.

٠٣

إيه اللى هيتطبع؟ خد بالك من علامتى التنصيص

اختيار من متعدد 
let total = 8;
total = total + "2";
console.log(total);

الإجابة الصح هى ب

الناتج82

السطر شكله تراكم عادى، بس الـ"2" بين تنصيص — يعنى نص مش رقم.

والعلامة + لما تلاقى نصاً على أى طرف بتوقف عن الجمع وتبدأ تلزّق: الرقم ٨ والنص "2" بقوا "82".

الشرح خطوة بخطوة
٠١

العلامة بتسأل عن النوع الأول

total + "2"

لو كان الطرفان رقمين كان طلع ١٠. طرف واحد نص كفاية إن الناتج يقلب.

٠٢

والناتج بيتخزّن فى الصندوق

total = total + "2";

وده أخطر جزء: الصندوق نفسه بقى نوعه نص. أى حساب جاى عليه هيتصرّف بنفس المنطق.

٠٤

إيه اللى هيتطبع؟

اختيار من متعدد 
let score = 90;
let grade = "A";
let passed = true;
console.log(grade);
console.log(passed);

الإجابة الصح هى د

الناتجA true

علامتا التنصيص بنكتبهما فى الكود عشان نقول للكمبيوتر «دى كلمة»، لكنهما مش بيتطبعا.

وtrue بتتكتب من غير تنصيص أصلاً — دى قيمة منطقية، نوع تالت غير الرقم والنص.

الشرح خطوة بخطوة
٠١

تلات أنواع فى تلات صناديق

let score = 90;
let grade = "A";
let passed = true;

الرقم بيتكتب من غير أى حاجة حواليه. والنص بين علامتى تنصيص. والقيمة المنطقية ليها احتمالان بس: true أو false، من غير تنصيص.

٠٢

الطباعة بتشيل التنصيص

console.log(grade);

العلامتان جزء من طريقة كتابة النص فى الكود، مش جزء من النص نفسه.

٠٥

إيه اللى هيتطبع؟

اختيار من متعدد 
console.log(typeof 7);
console.log(typeof "7");
console.log(typeof false);

الإجابة الصح هى ج

الناتجnumber string boolean

typeof معناها «إيه نوع الحاجة دى». بتكتبها وتسيب مسافة وتحط اللى عايز تسأل عنه — من غير أقواس.

ودى أول أداة فى حياتك بتسأل بيها الكمبيوتر سؤالاً بدل ما تأمره بأمر.

الشرح خطوة بخطوة
٠١

نفس الرقم، نوعان مختلفان

console.log(typeof 7);
console.log(typeof "7");

العلامتان بتغيّرا النوع، مش الشكل. الأول number والتانى string.

٠٢

والقيمة المنطقية نوع تالت

console.log(typeof false);

لو كتبت "false" بين تنصيص، بتبقى نصاً عادياً مش قيمة منطقية. جرّبها فى المحرّر تحت.

٠٦

الصندوق اتغيّر — طيب نوعه اتغيّر معاه؟

اختيار من متعدد 
let box = 12;
box = "12";
console.log(typeof box);

الإجابة الصح هى و

الناتجstring

الصندوق مش مقيّد بنوع. أول ما تحطّ فيه قيمة جديدة، نوعه بيبقى نوع القيمة الجديدة.

فى الأول كان فيه رقم، وبعد السطر التانى بقى فيه نص — وtypeof بتبصّ على اللى جواه دلوقتى.

الشرح خطوة بخطوة
٠١

إعادة الإسناد بتغيّر القيمة والنوع

box = "12";

الشكل المطبوع هيفضل 12 فى الحالتين — عشان كده الفرق ده صعب تلاحظه من غير typeof.

٠٢

وtypeof بتقول لك الحقيقة

console.log(typeof box);

لما تحتار فى أى تمرين، اكتب typeof قدام الحاجة وشوف. دى بوصلتك.

٠٧

تلات أسطر على نفس الصندوق — إيه اللى هيتطبع؟

اختيار من متعدد 
let n = "20";
console.log(n + 1);
console.log(Number(n) + 1);
console.log(n * 2);

الإجابة الصح هى ج

الناتج201 21 40

السطر الأول لزق: "20" نص فالعلامة لزقت الواحد جنبه.

والتانى حوّلنا الأول فبقى جمعاً: ٢١.

والتالت مفاجأة: علامة الضرب ملهاش معنى نصى أصلاً، فالكمبيوتر حوّل النص لرقم لوحده وضرب. وده الفرق: + بس هى اللى ليها معنيان.

الشرح خطوة بخطوة
٠١

علامة الزائد بتلزّق

console.log(n + 1);

العلامة + بتسأل عن النوع الأول. لقت نصاً، فاختارت اللزق.

٠٢

والتحويل بيرجّعها للجمع

console.log(Number(n) + 1);

خد بالك: Number(n) ما غيّرتش الصندوق. لو عايز تغيّره فعلاً، اكتب n = Number(n);

٠٣

وعلامة الضرب بتحوّل لوحدها

console.log(n * 2);

علامة + هى الوحيدة اللى ليها معنيان. باقى العلامات (- و* و/) ملهاش معنى نصى، فبتحوّل تلقائياً.

٠٨

رقمان بيتجمعوا ولا بيتلزقوا؟

اختيار من متعدد 
let a = 6;
let b = 7;
console.log(a + b);
console.log(String(a) + b);
console.log(String(a) + String(b));

الإجابة الصح هى أ

الناتج13 67 67

السطر الأول رقمان، فالعلامة جمعت: ١٣.

والتانى: حوّلنا واحداً بس لنص — وطرف واحد نص كفاية عشان العلامة تسيب الجمع وتلزّق.

والتالت حوّلنا الاتنين، والناتج نفسه — لأن اللزق حصل خلاص من أول ما لقى نصاً واحداً.

الشرح خطوة بخطوة
٠١

رقمان: جمع عادى

console.log(a + b);

مفيش نص فى أى طرف، فالعلامة بتشتغل بمعناها الحسابى.

٠٢

واحد بس بقى نصاً — وده كفاية

console.log(String(a) + b);

Number() بتفتح باب الحساب، وString() بتقفله وتفتح باب اللزق.

٠٣

والتانى ما زادش حاجة

console.log(String(a) + String(b));

الدرس: مش لازم تحوّل الطرفين. لو عايز لزقاً، حوّل واحداً وخلاص.

٠٩

مكان القوس بيغيّر كل حاجة

اختيار من متعدد 
console.log(Number("12") + Number("8"));
console.log(Number("12" + "8"));

الإجابة الصح هى د

الناتج20 128

السطر الأول: كل نص اتحوّل لوحده، فبقى ١٢ + ٨ = ٢٠.

والتانى: اللزق حصل جوه القوس الأول — "12" + "8" بقوا النص "128" — وبعدين اتحوّل كله لرقم واحد.

نفس الحروف بالظبط، وقوسان مكانهما مختلف، وناتجان مختلفان تماماً.

الشرح خطوة بخطوة
٠١

الأول: حوّل كل واحد لوحده

Number("12") + Number("8")

كل Number() اشتغلت على نصها لوحده، فبقى عندنا رقمان — والجمع بينهم عادى.

٠٢

والتانى: اللزق حصل جوه القوس

"12" + "8"

القاعدة القديمة: اللى جوه القوس بيتحسب الأول. وهنا اللى جواه نصان، فاللزق سبق التحويل.

٠٣

وبعدين التحويل شاف نصاً واحداً

console.log(Number("128"));

الدرس: بصّ على القوس قبل ما تحكم. مكانه بيحدّد إيه اللى بيحصل الأول.

١٠

إيه اللى هيتطبع؟ فكّر كويس

اختيار من متعدد 
let x = Number("abc");
console.log(x + 10);

الإجابة الصح هى ج

الناتجNaN

Number("abc") رجّعت NaN، وحُطّت فى الصندوق.

وأى عملية حسابية على NaN بترجّع NaN تانى. عشان كده لو دخلت فى برنامجك، بتنتشر لآخره وإنت مش هتلاقى سطر خطأ يقول لك إيه حصل.

الشرح خطوة بخطوة
٠١

التحويل فشل والناتج اتخزّن

let x = Number("abc");

الكود اشتغل عادى ومكمّل. المشكلة إنه شغّال بقيمة معطوبة.

٠٢

وأى حساب عليها بيرجّعها تانى

console.log(x + 10);

دى الحاجة الوحيدة فى البرمجة اللى ما بتساويش نفسها. ولو وصلت لك من غير ما تعرف منين، ارجع لأول Number() فى الكود.

١١

تراكم مع تحويل — إيه اللى هيتطبع؟

اختيار من متعدد 
let sum = "5";
sum = Number(sum) + 5;
sum = sum + 5;
console.log(sum);

الإجابة الصح هى ج

الناتج15

السطر التانى حوّل النص لرقم وخزّن الناتج فى الصندوق — فالصندوق بقى فيه الرقم ١٠.

وبعد كده السطر التالت بقى جمعاً عادياً بين رقمين: ١٠ + ٥ = ١٥.

الشرح خطوة بخطوة
٠١

التحويل مع الإسناد بيغيّر الصندوق فعلاً

sum = Number(sum) + 5;

الفرق عن تمرين ٧: هنا فيه علامة =، فالناتج المحوّل رجع للصندوق. من غيرها كان التحويل هيضيع.

٠٢

والباقى تراكم عادى

sum = sum + 5;

لو كنت شيلت Number() من السطر التانى، الصندوق كان هيفضل نصاً والناتج كان هيطلع 555.

١٢

قارن القيمة، وقارن النوع نفسه

اختيار من متعدد 
let a = "5";
let b = 5;
console.log(a == b);
console.log(a === b);
console.log(typeof a === typeof b);

الإجابة الصح هى و

الناتجtrue false false

علامتا يساوى == بتتساهل: بتحوّل قبل ما تقارن، فبتشوف الخمسة زى الخمسة.

وثلاث علامات === بتتشدّد: القيمة والنوع لازم يتطابقا.

والسطر التالت بيقارن الأنواع نفسها: "string" مع "number" — كلمتان مختلفتان، فالجواب false.

الشرح خطوة بخطوة
٠١

نفس القيمتين، علامتان مختلفتان

console.log(a == b);
console.log(a === b);

القيمة واحدة فى الحالتين. اللى بيفرق هو: هل العلامة بتسأل عن النوع كمان ولا لأ.

٠٢

والسطر التالت بيقارن الأنواع نفسها

console.log(typeof a === typeof b);

typeof بترجّع نصاً فيه اسم النوع. فالسطر ده بيقارن نصين: "string" و"number" — مختلفان، فالجواب false.

١٣

إيه اللى هيتطبع؟

اختيار من متعدد 
let a = 10;
let b = 3;
console.log(a > b);
console.log(a < b);
console.log(typeof (a > b));

الإجابة الصح هى هـ

الناتجtrue false boolean

علامات المقارنة بترجّع قيمة منطقية: true أو false — مش رقماً ومش نصاً.

ودى القيم اللى الشروط هتشتغل بيها فى المحاضرة الجاية، فخلّى بالك منها.

الشرح خطوة بخطوة
٠١

المقارنة بترجّع نوعاً تالتاً

console.log(a > b);

الكمبيوتر ما بيرجّعش «آه» أو «لأ» — بيرجّع true أو false.

٠٢

وtypeof بتأكّد لك

console.log(typeof (a > b));

الأقواس حوالين a > b عشان الكمبيوتر يحسب المقارنة الأول وبعدين يسأل عن نوع الناتج.

١٤

اسمان شبه بعض — ولا علاقة بينهما

اختيار من متعدد 
let my_score = 4;
let myScore = 10;

الإجابة الصح هى ب

الناتج15

فدول صندوقان منفصلان: الأول بقى ٥ بعد التراكم، والتانى ١٠، والمجموع ١٥.

الشرح خطوة بخطوة
٠١

صندوقان منفصلان رغم التشابه

let my_score = 4;
let myScore = 10;

الكمبيوتر ما بيخمّنش إن الاسمين يقصدوا نفس الحاجة. أى فرق حرف واحد = اسم تانى خالص.

٠٢

والتراكم بيلمس واحداً بس

my_score = my_score + 1;

وفى الآخر ٥ + ١٠ = ١٥. العُرف المتبع فى JavaScript هو myScore — اختار واحداً والتزم بيه.

١٥

التمرين الأصعب فى القسم ده

الإجابة الصح هى ب

الناتج30 number

السطر التانى لزق: الرقم ٣ مع النص "0" بقوا النص "30".

الشرح خطوة بخطوة
٠١

اللزق بيعمل نصاً جديداً

v = v + "0";

لحد هنا الصندوق فيه نص مكوّن من حرفين، مش الرقم ٣٠.

v = Number(v);

١٦

اكتب قيمة كل صندوق بعد كل سطر

الإجابة الصح هى ج

الناتج1 7

السطر التالت نقل نسخة من p (اللى هى ٧) لـq.

الشرح خطوة بخطوة
٠١

النسخ بياخد القيمة اللحظية

القيمة القديمة بتاعة q (اللى كانت ٢) ضاعت هنا.

والتغيير بعده مالوش أثر رجعى

١٧

فين الغلط؟ فيه غلطتان

اختيار من متعدد 
let 2nd = 5;
let name = "Ali";
console.log(Name);

الإجابة الصح هى هـ

let second = 5;
let name = "Ali";
console.log(name);

الغلط الأول — اسم الصندوق 2nd بيبدأ برقم، وده ممنوع. الاسم لازم يبدأ بحرف أو شرطة سفلية. الصح مثلاً: second.

الغلط التانى — الصندوق اسمه name بحرف صغير، واحنا طلبنا Name بحرف كبير. دول اسمان مختلفان تماماً.

الشرح خطوة بخطوة
٠١

الاسم ما يبدأش برقم

let 2nd = 5;      // غلط
let second = 5;   // صح

المسموح فى الاسم: حروف وأرقام وشرطة سفلية — بشرط ما يبدأش برقم.

٠٢

console.log(Name);   // غلط
console.log(name);   // صح

دى نفس القاعدة اللى هتقابلك مع Number() فى المحاضرة دى — الحرف الكبير مهم.

أكمل الناقص

اختيار من متعدد 
let items = 9;
let each = 5;
let total = items * each;
console.log(______ + total);

الإجابة الصح هى أ

let items = 9;
let each = 5;
let total = items * each;
console.log("Total = " + total);
الناتجTotal = 45

الفراغ مكانه نص بين علامتى تنصيص: "Total = ".

وخد بالك من المسافة قبل علامة التنصيص اللى بتقفل — من غيرها هيطلع Total =45 ملزوقة.

الشرح خطوة بخطوة
٠١

الحساب الأول

let total = items * each;

لحد هنا total فيه الرقم ٤٥.

٠٢

واللزق بيوصّل النص بالرقم

console.log("Total = " + total);

التحويل من رقم لنص بيحصل تلقائياً هنا — مش محتاج String().

٢٠

اكتب النوع بعد كل سطر

اختيار من متعدد 
let a = 3.5;
let b = "";
let c = false;
console.log(typeof a);
console.log(typeof b);
console.log(typeof c);

الإجابة الصح هى د

الناتجnumber string boolean

الكسر العشرى 3.5 نوعه number برضه — مفيش نوع منفصل للكسور.

والنص الفاضى "" لسه نصاً — فيه علامتا تنصيص، وده كفاية.

وfalse نوعها boolean زى true بالظبط.

الشرح خطوة بخطوة
٠١

الكسر رقم برضه

console.log(typeof 3.5);

ولاحظ إن الفاصلة العشرية بتتكتب نقطة فى الكود: 3.5 مش 3,5.

٠٢

والنص الفاضى لسه نص

console.log(typeof "");

النوع بيتحدّد من شكل الكتابة، مش من المحتوى.

٢١

اكتب الأربع نواتج — وقول ليه

اختيار من متعدد 
console.log(2 == "2");
console.log(2 === "2");
console.log(2 === 2);
console.log("2" === "2");

الإجابة الصح هى هـ

الناتجtrue false true true

علامتا يساوى == بتحوّل قبل ما تقارن، فبتشوف القيمتين واحدة.

وثلاث علامات === بتقارن القيمة والنوع — فالسطر التانى بيطلع false لأن واحد رقم والتانى نص.

والسطران الأخيران true لأن كل واحد بيقارن نوعاً بنفس النوع.

الشرح خطوة بخطوة
٠١

المتساهلة والمتشدّدة

console.log(2 == "2");
console.log(2 === "2");

== بتحوّل النص لرقم وبعدين تقارن. و=== بتشوف النوعين مختلفين فبترفض من غير ما تحوّل.

٠٢

ولما النوعان يتطابقا

console.log("2" === "2");

نصان متطابقان حرفاً بحرف = true بالعلامتين. النصيحة: استعمل === دايماً.

٢٢

فين الغلط؟

اختيار من متعدد 
let s = "30";
let n = number(s);
console.log(n + 1);

الإجابة الصح هى و

let s = "30";
let n = Number(s);
console.log(n + 1);
الناتج31

الغلط فى number(s) بحرف n صغير. الاسم الصح Number بحرف N كبير.

والكمبيوتر حساس لحالة الحرف — زى ما شفنا فى أسماء الصناديق بالظبط. والتصحيح بيدّى 31.

الشرح خطوة بخطوة
٠١

الحرف الكبير مش تفصيلة

let n = number(s);   // غلط
let n = Number(s);   // صح

نفس القاعدة بتنطبق على String() — بحرف S كبير.

٠٢

وبعد التصحيح

let n = Number(s);
console.log(n + 1);

من غير التحويل كان الناتج هيطلع 301 بالزق.

٢٣

من فين جت NaN؟

اختيار من متعدد 
let a = 5;
let b = Number("5 كتب");
let c = a + b;
console.log(c);

الإجابة الصح هى ب

الناتجNaN

السطر التانى هو المسؤول: النص "5 كتب" مش رقم كامل — فيه حروف — فالتحويل فشل ورجّع NaN.

والسطر التالت جمع ٥ على NaN، وأى حساب على NaN بيرجّع NaN. فالمشكلة ظهرت فى السطر الرابع، لكن مصدرها التانى.

الشرح خطوة بخطوة
٠١

التحويل فشل هنا

let b = Number("5 كتب");

مش شرط النص يكون كله حروفاً. أى حرف واحد جواه بيكفى إن التحويل يفشل.

٠٢

وبعدين انتشرت

let c = a + b;
console.log(c);

الدرس العملى: لما تشوف NaN، ما تدوّرش على السطر اللى طبعها — دوّر على أول Number() فى الكود.

٢٤

أقل تعديل — عايزين الناتج ٨

اختيار من متعدد 
let x = "5";
let y = 3;
console.log(x + y);

الإجابة الصح هى هـ

let x = 5;
let y = 3;
console.log(x + y);
الناتج الحالى53

دلوقتى بيطبع 53 — لزق، لأن x نص.

وفيه حلّان صحيحان: تشيل علامتى التنصيص من "5"، أو تكتب Number(x) + y. الاتنين بيدّوا ٨.

والأول أنضف لو الرقم من عندك. والتانى هو الحل الوحيد لو النص جاى من برّه ومش متحكّم فيه.

الناتج بعد التعديل8
الشرح خطوة بخطوة
٠١

دلوقتى: نص مع رقم

console.log(x + y);

العلامة شافت نصاً فاختارت اللزق. عايزين نودّيها للفرع التانى.

٠٢

الحل الأول: شيل التنصيص

let x = 5;   // من غير تنصيص

بقوا رقمين، فالعلامة رجعت لمعناها الحسابى.

٠٣

أو الحل التانى: حوّل وقت الحساب

console.log(Number(x) + y);

استعمل ده لما ما تكونش متحكّماً فى مصدر النص.

٢٥

اكتب كوداً — اسأل عن النوع

اختيار من متعدد

الإجابة الصح هى أ

let v = "true";
console.log(v);
console.log(typeof v);
الناتجtrue string

الطباعة بتطلّع true — من غير التنصيص، زى أى نص.

لكن النوع string مش boolean، لأن العلامتين موجودتان فى الكود. القيمة المنطقية بتتكتب true من غير أى علامات.

وده أخطر التباس فى المحاضرة: الشكل المطبوع واحد، والنوع مختلف تماماً.

الشرح خطوة بخطوة
٠١

الطباعة بتخفى الفرق

console.log(v);

لو اعتمدت على الطباعة وبس، مستحيل تفرّق بينهم.

٠٢

وtypeof بتفضحه

console.log(typeof v);

لما تحتار، اكتب typeof قدام الحاجة وشوف. دى بوصلتك فى المحاضرة دى كلها.

٢٦

اكتب الأربع نواتج

اختيار من متعدد 
console.log(10 > 3);
console.log(10 < 3);
console.log("10" > 3);
console.log(typeof (10 > 3));

الإجابة الصح هى ج

الناتجtrue false true boolean

أول سطرين مقارنة عادية بين رقمين.

والسطر التالت مفاجأة: "10" نص، لكن علامة > ملهاش معنى نصى للمقارنة العددية، فالكمبيوتر حوّل النص لرقم لوحده وقارن — زى ما عملت علامة الضرب بالظبط.

والأخير: ناتج أى مقارنة نوعه boolean دايماً.

الشرح خطوة بخطوة
٠١

المقارنة بترجّع قيمة منطقية

console.log(10 > 3);

الكمبيوتر ما بيرجّعش «آه» أو «لأ» — بيرجّع true أو false.

٠٢

وعلامات المقارنة بتحوّل لوحدها

console.log("10" > 3);

علامة + هى الاستثناء الوحيد. كل العلامات التانية (- * / < >) بتحوّل لوحدها.

٠٣

ونوع الناتج ثابت

console.log(typeof (10 > 3));

الأقواس ضرورية: من غيرها الكمبيوتر هيسأل عن نوع 10 بس وبعدين يقارن.

٢٧

اكتب القيمة والنوع بعد كل سطر

اختيار من متعدد 
let t = 4;         // (1)
t = t + 1;         // (2)
t = t + "0";       // (3)
t = Number(t);     // (4)
console.log(t);
console.log(typeof t);

الإجابة الصح هى د

الناتج50 number

(١) ٤ — number. (٢) ٥ — number. (٣) "50" — string. (٤) ٥٠ — number.

السطر (٣) هو المفتاح: اللزق حوّل الصندوق كله لنص، والرقم بقى "50" بحرفين.

والسطر (٤) رجّعه رقماً — والنص كان أرقاماً فعلاً فالتحويل نجح.

الشرح خطوة بخطوة
٠١

السطر (٣): اللزق غيّر النوع

t = t + "0";

صفر واحد بين تنصيص كفاية إن الصندوق كله يقلب نوعه.

٠٢

والسطر (٤): التحويل رجّعه

t = Number(t);

لاحظ إن فيه علامة = فى السطر ده — عشان كده الصندوق نفسه اتغيّر. من غيرها كان التحويل هيضيع.

٠٣

والطباعة ما بتقولش لك النوع

console.log(t);
console.log(typeof t);

لو ما كانش السطر (٤) موجود، الطباعة كانت هتطلّع 50 برضه — بس النوع كان string.

٢٨

التمرين الختامى — تراكم بيقلب نوعه

اختيار من متعدد 
let g = 0;
g = g + 5;
g = g + "5";
g = g + 5;
console.log(g);

الإجابة الصح هى و

الناتج555

السطر التانى جمع عادى: ٠ + ٥ = ٥.

والسطر التالت هو اللى قلبه: الرقم ٥ مع النص "5" بقوا النص "55".

وبعد كده الصندوق بقى نصاً، فالسطر الرابع لزق برضه: "55" مع ٥ بقوا "555". سطر واحد فيه تنصيص خرّب اللى بعده كمان.

الشرح خطوة بخطوة
٠١

لحد السطر التانى: أرقام

g = g + 5;

تراكم عادى تماماً، والصندوق نوعه number.

٠٢

السطر التالت: نقطة الانقلاب

g = g + "5";

هنا بالظبط. الصندوق كان رقماً وبقى نصاً، وده هيأثّر على كل سطر بعده.

٠٣

والسطر الرابع اتأثّر

g = g + 5;   // نفس السطر التانى بالحرف

ده أهم درس فى الورقة: النوع بيسرى للأمام. غلطة واحدة فى سطر بتغيّر معنى كل سطر بعده.

٢٩

المشروع الأول — فاتورة

كتابة

اكتب برنامجاً كاملاً بالمواصفات دى: ١ — صندوق item فيه اسم الصنف "Pen". ٢ — صندوق price فيه النص "12" (زى ما بييجى من خانة كتب فيها المستخدم). ٣ — صندوق count فيه الرقم ٣. ٤ — صندوق total فيه ناتج ضرب السعر فى العدد كرقم — خد بالك إن السعر نص. ٥ — اطبع نوع total عشان تتأكد إنه number. ٦ — اطبع سطراً بالشكل ده بالظبط: Pen x3 = 36

اتحفظت 
let item = "Pen";
let price = "12";
let count = 3;
let total = Number(price) * count;
console.log(typeof total);
console.log(item + " x" + count + " = " + total);
الناتجnumber Pen x3 = 36

المفتاح فى السطر الرابع: price نص، فلازم Number() قبل الضرب.

ولو كتبت price * count من غير تحويل، هتلاقى الناتج ٣٦ برضه — لأن علامة الضرب بتحوّل لوحدها. بس typeof هتقول لك number فى الحالتين، فالفرق مش هيبان هنا.

الفرق بيبان لو استعملت + بدل *: ساعتها التحويل بيبقى إجبارى.

وفى السطر الأخير: اللزق بيوصّل النص بالرقم تلقائياً، والمسافات اللى جوه التنصيص بتتطبع زى ما هى.

الشرح خطوة بخطوة
٠١

السعر بييجى نصاً — ده الوضع الطبيعى

let price = "12";

دى مش حالة استثنائية — دى الحالة الافتراضية فى أى برنامج حقيقى بياخد مدخلات.

٠٢

التحويل قبل الحساب

let total = Number(price) * count;

اكتب Number() حتى لو العلامة هتحوّل لوحدها — عشان اللى بيقرا كودك يفهم نيّتك.

٠٣

والطباعة المنسّقة بتلزّق

console.log(item + " x" + count + " = " + total);

عدّ المسافات: " x" فيها مسافة قبل، و" = " فيها مسافة قبل وبعد. من غيرهم الناتج هيطلع ملزوقاً.

٣٠

المشروع التانى — رصيد بيتغيّر

كتابة

اكتب برنامجاً كاملاً بالمواصفات دى: ١ — صندوق balance فيه الرقم ١٠٠. ٢ — اخصم منه ٣٠ (تراكم). ٣ — ضيف له النص "20" من غير تحويل، واطبع القيمة والنوع. ٤ — بعد كده صلّح الوضع: رجّع balance رقماً بـNumber(). ٥ — اطبع القيمة والنوع تانى. ٦ — وفى الآخر اطبع: Final = 7020

اتحفظت 
let balance = 100;
balance = balance - 30;
balance = balance + "20";
console.log(balance);
console.log(typeof balance);
balance = Number(balance);
console.log(balance);
console.log(typeof balance);
console.log("Final = " + balance);
الناتج7020 string 7020 number Final = 7020

السطر التالت هو نقطة الانقلاب: 70 رقم، و"20" نص، فالعلامة + لزقت وطلعت "7020" — وده نص مش رقم.

والسطر السادس رجّعه رقماً: النص "7020" كله أرقام فالتحويل نجح.

خد بالك إن القيمة المطبوعة 7020 واحدة قبل التحويل وبعده — الشكل ما اتغيّرش، النوع بس هو اللى اتغيّر. وtypeof هى الوحيدة اللى بتفضح الفرق.

الدرس: لو ما كتبتش typeof، كنت هتفتكر إن البرنامج شغّال صح — وهو شغّال بنوع غلط.

الشرح خطوة بخطوة
٠١

التراكم عادى لحد هنا

balance = balance - 30;

الصندوق لسه number، وكل حاجة طبيعية.

٠٢

وهنا الانقلاب

balance = balance + "20";

لو كان 20 من غير تنصيص كان الناتج ٩٠. علامتان قلبتا الحساب لِلَزق.

٠٣

والتصحيح بيرجّع النوع

balance = Number(balance);

لاحظ علامة = فى السطر ده — من غيرها التحويل كان هيضيع والصندوق كان هيفضل نصاً.
