القاعدة الوحيدة

جاوب الأول، وبعدين اكشف. لو كشفت قبل ما تفكّر، التمرين بيبقى ملوش أى لازمة — إنت بتخدع نفسك مش المدرس.

و«الصندوق» هنا هو نفسه «الدرج» بتاع المحاضرة، وهو نفسه اللى الكتب بتسمّيه متغير.

الناتج7 * 6 42

اللى بين علامتى التنصيص نص، بيتطبع حرف بحرف زى ما هو. واللى من غيرهما عملية حسابية، بتتحلّ الأول وبعدين الناتج بيتطبع.

٠٢

السطر التانى: من غير تنصيص، يبقى حساب

console.log(7 * 6);

علامتان صغيرتان بتقلبا معنى السطر كله: من كلمة مطبوعة إلى عملية محسوبة.

الناتج10 10

الشكل المطبوع واحد، بس الحاجتين مش نفس الحاجة.

الأول نص مكوّن من حرفين، والتانى رقم تقدر تجمعه وتضربه. والفرق ده هيبان فى آخر تمرينين فى الورقة دى.

هنا بس بيبان الفرق: الأول بيطلع 105 والتانى بيطلع 15. جرّبهما بنفسك.

٠٣

والسطر التانى من غير تنصيص، فالكمبيوتر فهم إن Ahmed اسم صندوق، وراح يدوّر عليه فما لقاهوش — لأن محدش عمله بـlet.

الشرح خطوة بخطوة
٠١

السطر الأول: كلمة عادية

console.log("Ahmed");

بين تنصيص، فمفيش أى سؤال — اطبعها وخلاص.

٠٢

السطر التانى: الكمبيوتر بيدوّر على صندوق

console.log(Ahmed);

ترجمة الرسالة: «Ahmed ده مش معرّف عندى». وده أشهر خطأ بيقابلك أول أسبوع، وعلاجه سطر واحد: let Ahmed = "Ahmed";

٠٤

السطر التالت هو أخطر واحد: رقم واحد بس بين تنصيص كفاية إن النتيجة تقلب من ٧ لـ٣٤.

الشرح خطوة بخطوة
٠١

نص مع نص: لزق

console.log("3" + "4");

زى ما تلزّق ورقتين ورا بعض: الحرف ٣ وبعده الحرف ٤.

٠٢

رقم مع رقم: جمع

console.log(3 + 4);

هنا العلامة رجعت لمعناها الحسابى العادى.

٠٣

نص واحد كفاية

console.log("3" + 4);

مش لازم الاتنين يكونوا نصاً. واحد بس كفاية عشان العلامة تسيب الجمع وتلزّق.

٠٥

اختيار من متعدد 
let city;
console.log(city);

console.log(city);

الإجابة الصح هى هـ

الناتجundefined Cairo

let city; عمل الصندوق بس ساب جواه فاضى، فلما طلبنا نطبعه قال undefined — يعنى «الصندوق موجود بس مفيش حاجة جواه لسه».

وبعدين السطر التالت حطّ "Cairo" جواه، من غير let لأن الصندوق اتعمل خلاص.

الشرح خطوة بخطوة
٠١

السطر الأول: صندوق من غير حاجة جواه

let city;

الصندوق اتعمل واتكتب عليه الاسم، بس محدش حطّ فيه حاجة لسه.

٠٢

الطباعة بتقول لك إنه فاضى

console.log(city);

كلمة undefined مش خطأ. دى طريقة الكمبيوتر إنه يقول لك «الصندوق موجود، بس فاضى».

٠٣

دلوقتى بس بنملاه

city = "Cairo";
console.log(city);

خد بالك: مفيش let فى السطر ده. الصندوق موجود خلاص، واحنا بس بنملاه.

٠٦

إيه اللى هيتطبع؟ الفرق فى علامتى التنصيص

اختيار من متعدد 
let n = 7;
console.log(n);
console.log("n");

الإجابة الصح هى ج

الناتج7 n

n من غير تنصيص معناها «افتح الصندوق اللى اسمه n واطبع اللى جواه».

و"n" بين تنصيص معناها «اطبع الحرف n» — من غير ما تبص على أى صندوق.

الشرح خطوة بخطوة
٠١

من غير تنصيص: افتح الصندوق

console.log(n);

الكمبيوتر فهم إن n اسم صندوق، فراح فتحه وطبع اللى لقاه.

٠٢

بين تنصيص: اطبع الحرف

console.log("n");

هنا ما بصّش على أى صندوق. شايف حرفاً بين تنصيص، فطبعه.

٠٧

حرف واحد كبير — إيه اللى هيحصل؟

اختيار من متعدد 
let studentName = "Omar";
console.log(StudentName);

الإجابة الصح هى و

الناتجReferenceError: StudentName is not defined

الصندوق اسمه studentName بحرف s صغير، واحنا طلبنا StudentName بحرف كبير.

دول اسمان مختلفان تماماً عند الكمبيوتر — الأسماء حساسة لحالة الحرف. والتصحيح: console.log(studentName);

الشرح خطوة بخطوة
٠١

الصندوق اتعمل باسم معيّن

let studentName = "Omar";

الاسم المكتوب على الصندوق هو studentName بالظبط، بحروفه الكبيرة والصغيرة زى ما هى.

٠٢

طلبنا اسماً تانى

console.log(StudentName);

الكمبيوتر ما بيخمّنش. S غير s، فهو شايف اسمين مختلفين تماماً.

٠٨

إيه اللى هيتطبع؟

اختيار من متعدد 
let color = "red";
color = "green";
color = "blue";
console.log(color);

الإجابة الصح هى ج

الناتجblue

الصندوق بياخد حاجة واحدة بس، وكل قيمة جديدة بتشيل اللى قبلها خالص. والطباعة فى آخر سطر، فهى بتشوف آخر قيمة اتحطّت.

يعنى "red" و"green" ضاعوا خالص، ومفيش طريقة نرجّعهم بيها.

الشرح خطوة بخطوة
٠١

السطر الأول: الصندوق بيتعمل وبيتملى

let color = "red";

السطر ده عمل حاجتين مع بعض: عمل صندوق اسمه color، وحطّ جواه النص "red".

٠٢

السطر التانى: القيمة القديمة بتتشال

color = "green";

مفيش let فى السطر ده، لأن الصندوق اتعمل خلاص. واللى حصل خطوتين: "red" اتشالت، و"green" اتحطّت مكانها.

٠٣

السطر التالت: نفس الحكاية تانى

color = "blue";

عشان كده اسمه «متغير»: الاسم ثابت، واللى جواه بيتغيّر كل ما تحطّ قيمة جديدة.

٠٤

السطر الأخير: الطباعة بتفتح الصندوق

console.log(color);

الطباعة بتشوف اللى فى الصندوق فى اللحظة دى بس. لو حطّيتها فى أول سطر كانت هتطبع red.

٠٩

صندوقين — واحد بياخد نسخة من التانى

اختيار من متعدد 
let x = 4;
let y = x;
x = 9;
console.log(x);
console.log(y);

الإجابة الصح هى أ

الناتج9 4

السطر let y = x; ما ربطش الصندوقين ببعض. هو بصّ على اللى جوه x فى اللحظة دى (٤)، وحطّ نسخة منه فى y.

وبعد كده أى تغيير فى x مالوش أى علاقة بـy. الصورة الصح: نقلت ورقة من صندوق لصندوق، مش وصّلت الصندوقين بماسورة.

الشرح خطوة بخطوة
٠١

السطر الأول: صندوق واحد بس

let x = 4;

لحد هنا كله عادى: صندوق اتعمل، واتحطّ جواه ٤.

٠٢

السطر التانى: نسخة، مش وصلة

let y = x;

دى أهم نقطة فى التمرين: النسخ حصل مرة واحدة بس، فى اللحظة دى. وبعد كده كل صندوق بقى لوحده.

٠٣

السطر التالت: x بس هو اللى اتغيّر

x = 9;

الغلطة اللى بيقع فيها ناس كتير إنهم يفتكروا إن y هيتغيّر مع x. لو ده كان بيحصل، كان الناتج طلع ٩ و٩.

٠٤

آخر سطرين: كل صندوق بيطبع اللى جواه

console.log(x);
console.log(y);

فالناتج بيطلع سطرين: 9 وبعده 4.

١٠

إيه اللى هيتطبع؟ فكّر كويس قبل ما تجاوب

اختيار من متعدد 
let total = 5;
total + 3;
console.log(total);

الإجابة الصح هى ب

الناتج5

السطر التانى حسب ٥ + ٣ فعلاً وطلع ٨ — بس رماه فى الهوا.

مفيش علامة =، يعنى مفيش حد قال «حُطّ الناتج فين». والتصحيح: total = total + 3; وساعتها يطبع ٨.

الشرح خطوة بخطوة
٠١

السطر التانى بيحسب فعلاً

total + 3;

الكمبيوتر فعلاً حسب. المشكلة مش فى الحساب.

٠٢

بس محدش قال له يحطّه فين

total + 3;      // الناتج ضاع
total = total + 3;  // الناتج اتخزّن

القاعدة: أى حساب من غير إسناد بيضيع. علامة = هى اللى بتقول «خُد الناتج ده وحُطّه هنا».

١١

إيه اللى هيتطبع؟

اختيار من متعدد 
let points = 0;
points = points + 5;
points = points + 5;
points = points + 10;
console.log(points);

الإجابة الصح هى د

الناتج20

القاعدة كل مرة واحدة: احسب اللى على يمين العلامة = الأول، وبعدين حُطّ الناتج فى الصندوق.

يعنى ٠ ثم ٥ ثم ١٠ ثم ٢٠.

الشرح خطوة بخطوة
٠١

الخطوة الأولى: ٠ زائد ٥

points = points + 5;

شكله غلط رياضياً — مفيش رقم بيساوى نفسه زائد ٥. بس افتكر إن = معناها «حُطّ فى» مش «يساوى».

٠٢

الخطوة التانية: نفس السطر بقيمة جديدة

points = points + 5;

السطر مكتوب بنفس الحروف بالظبط، بس الناتج اتغيّر — لأن اللى فى الصندوق اتغيّر.

٠٣

الخطوة الأخيرة

points = points + 10;
console.log(points);

وفى الآخر الطباعة بتفتح الصندوق وتلاقى ٢٠.

١٢

تراكم بالضرب — إيه اللى هيتطبع؟

اختيار من متعدد 
let k = 2;
k = k * 3;
k = k * 3;
console.log(k);

الإجابة الصح هى أ

الناتج18

٢ ثم ٦ ثم ١٨.

نفس قاعدة التراكم بالظبط، بس بعلامة الضرب: اقرا اللى جوه الصندوق، اضربه فى ٣، رجّع الناتج للصندوق.

الشرح خطوة بخطوة
٠١

المرة الأولى: ٢ فى ٣

k = k * 3;

الصندوق فيه ٢، فالعملية بقت ٢ × ٣.

٠٢

المرة التانية: ٦ فى ٣

k = k * 3;

دلوقتى الصندوق فيه ٦، فنفس السطر بقى ٦ × ٣. لو حسبتها ٢ × ٣ × ٣ هتوصل لنفس الرقم بالصدفة — بس الطريقة اللى بتحصل فعلاً هى دى.

١٣

صندوقين بيتغيّرا بالترتيب

اختيار من متعدد 
let a = 6;
let b = 4;
a = a - b;
b = b - a;
console.log(a);
console.log(b);

الإجابة الصح هى هـ

الناتج2 2

السطر التالت: ٦ ناقص ٤ يساوى ٢، فـa بقى ٢.

والسطر الرابع بيقرا a بقيمته الجديدة: ٤ ناقص ٢ يساوى ٢. لو حسبته بقيمة a القديمة (٦) كنت هتطلع ناقص ٢ — وده الغلط الشائع.

الشرح خطوة بخطوة
٠١

السطر التالت: a بيتغيّر

a = a - b;

الصندوق b ما اتغيّرش هنا — احنا قرينا منه بس.

٠٢

السطر الرابع: بيقرا a الجديدة

b = b - a;   // a دلوقتى 2 مش 6

هنا الفخ: a بقى ٢ من السطر اللى فات. الكمبيوتر بيقرا القيمة الموجودة دلوقتى، مش القيمة الأصلية.

٠٣

الناتج

console.log(a);
console.log(b);

الكمبيوتر بينفّذ سطراً سطراً بالترتيب، وما بيرجعش لورا.

١٤

إيه اللى هيتطبع؟ خد بالك من المسافة

اختيار من متعدد

الإجابة الصح هى و

الناتجArea = 21

السطر التالت حسب اليمين الأول (٧ × ٣) وحطّ ٢١ فى صندوق جديد.

ولاحظ المسافة اللى سيبناها جوه التنصيص قبل ما يقفل: "Area = ". من غيرها كان هيطلع Area =21 ملزوقة.

الشرح خطوة بخطوة
٠١

الحساب الأول، والتخزين بعده

let area = width * height;

الصندوق area صندوق جديد تماماً، فمحتاج let زى أى صندوق جديد.

٠٢

اللزق بيوصّل النص بالرقم

console.log("Area = " + area);

المسافة قبل علامة التنصيص اللى بتقفل جزء من النص، فبتتطبع معاه.

١٥

نفس الصندوقين، علامتان مختلفتان

اختيار من متعدد 
let a = "10";
let b = 5;
console.log(a + b);
console.log(a - b);

الإجابة الصح هى ب

الناتج105 5

العلامة + ليها معنيان: جمع، ولزق. ولما تلاقى نصاً بتختار اللزق — فطلعت 105.

لكن العلامة - ملهاش معنى نصى أصلاً، فالكمبيوتر اضطر يعامل "10" كرقم وطرح عادى: ١٠ ناقص ٥ يساوى ٥.

الشرح خطوة بخطوة
٠١

علامة الزائد بتلزّق

console.log(a + b);

الحرفان 1 و0 وبعدهما 5. مفيش أى جمع حصل.

٠٢

علامة الناقص ملهاش معنى نصى

console.log(a - b);

ليه ده مهم دلوقتى: التمرين ده باب المحاضرة الجاية بالظبط — أنواع البيانات، وإمتى الكمبيوتر بيحوّل من نوع لنوع لوحده.

١٦

أكمل الناقص

اختيار من متعدد

الفراغ ده مكانه إيه عشان السطر الأخير يطبع 25؟

let price;
______ = 25;
console.log(price);

الإجابة الصح هى د

let price;
price = 25;
console.log(price);
الناتج25

الفراغ مكانه اسم الصندوق price لوحده، من غير let — لأن الصندوق اتعمل خلاص فى السطر اللى فوق.

ولو كتبت let price = 25; هيبقى غلط: عملت الصندوق مرتين.

الشرح خطوة بخطوة
٠١

السطر الأول عمل الصندوق وسابه فاضى

let price;

كلمة let اشتغلت خلاص. مهمتها إنها تعمل الصندوق، وهى عملته.

٠٢

الملء بيبقى بالاسم لوحده

price = 25;   // صح
let price = 25;  // غلط: الصندوق موجود

الطريقة اللى تفكّر بيها: let بتعمل الصندوق، والعلامة = بتملاه. الأولى مرة واحدة، والتانية زى ما تحب.

١٧

اكتب كوداً — أربعة أسطر

let pens = 12;

let بتعمل الصندوق، فبتتكتب مرة واحدة. وبعد كده أى تغيير بيبقى إسناداً عادياً باسم الصندوق والعلامة = بس.

الشرح خطوة بخطوة
٠١

أول سطرين: اعمل واطبع

let pens = 12;
console.log(pens);

الطباعة بتشوف القيمة الموجودة فى اللحظة دى.

١٨

فين الغلط؟ وإيه الناتج بعد التصحيح؟

اختيار من متعدد

المشروع التانى — السعر اللى جاى نصاً

كتابة

اكتب برنامجاً كاملاً بالمواصفات دى: ١ — صندوق price فيه النص "8" — بالتنصيص. ٢ — صندوق qty فيه الرقم ٢ — من غير تنصيص. ٣ — اطبع price + qty. ٤ — اطبع price - qty. ٥ — بعد كده صلّح الوضع: خلّى price رقماً ٨ من غير تنصيص، من غير ما تكتب let تانى. ٦ — اطبع price + qty تانى.

وفى السطر الخامس شيلنا علامتى التنصيص، فالصندوق بقى فيه رقم — والزائد رجع يجمع وطلع 10.
