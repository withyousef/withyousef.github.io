تحديد المستوى

خمستاشر سؤالاً فى ١٢ دقيقة. الورقة مش داخلة فى أى درجة، والهدف الوحيد منها إنك تعرف القاعة واقفة فين قبل ما تبدأ تبنى. الطباعة بتطلّع ورقة الطالب لوحدها، والمفتاح مستبعد تماماً.

ورقة الطالب

اختبار تحديد المستوى — ١٥ سؤالاً

إعداد: يوسف زكريا التهامى · تأسيس البرمجة · بكالوريا ٢٠٢٧
الاسمالتاريخ

ده مش امتحان ومش داخل فى أى درجة. ده مقياس بداية علشان أعرف أبدأ منين معاكم بالظبط. لو مش عارف إجابة سؤال، سيبه فاضى — الفراغ ده معلومة مفيدة ليا أكتر من تخمين. الوقت: ١٢ دقيقة.

القسم الأول — تتبّع الكود: إيه اللى هيتطبع؟

١) الكود ده هيطبع إيه؟

let fish = 0;
fish = fish + 3;
fish = fish + 3;
console.log(fish);
  • أ) 0
  • ب) 3
  • ج) 6
  • د) 33

٢) الكود ده هيطبع إيه؟

console.log("The answer is " + (2 * 5) + ".");
  • أ) The answer is 2 * 5.
  • ب) The answer is 10.
  • ج) The answer is 25.
  • د) رسالة خطأ

٣) الكود ده هيطبع إيه؟

let science = 70;
if (science == 100) {
  console.log("Perfect");
} else if (science >= 50) {
  console.log("Pass");
} else {
  console.log("Fail");
}
  • أ) Perfect
  • ب) Pass
  • ج) Fail
  • د) Pass ثم Fail

٤) الكود ده هيطبع إيه؟

for (let i = 1; i <= 3; i++) {
  console.log(i * 2);
}
  • أ) 1 ثم 2 ثم 3
  • ب) 2 ثم 4 ثم 6
  • ج) 2 ثم 4 ثم 6 ثم 8
  • د) 6

٥) الكود ده هيطبع إيه؟

let tempList = [36, 38, 26];
console.log(tempList[1] + tempList.length);
  • أ) 39
  • ب) 41
  • ج) 29
  • د) 363

القسم الثانى — إكمال الكود: إيه اللى ينزل مكان الفراغ؟

٦) علشان تعمل متغيراً اسمه score وتحط فيه القيمة ١٠، تكتب:

  • أ) let score = 10;
  • ب) score let = 10;
  • ج) let score == 10;
  • د) let = score 10;

٧) علشان الحلقة دى تلف خمس مرات بالظبط، الفراغ يتملى بإيه؟

for (let i = 0; i ______ 5; i++) { }
  • أ) >
  • ب) <
  • ج) <=
  • د) =

٨) علشان الشرط يتحقق لما يكون العمر أكبر من ١٢ و أصغر من ٢٠ فى نفس الوقت:

if (age > 12 ______ age < 20) { }
  • أ) ||
  • ب) &&
  • ج) +
  • د) ==

٩) علشان الدالة ترجّع الناتج بدل ما تطبعه:

function double(n) {
  ______ n * 2;
}
  • أ) console.log
  • ب) let
  • ج) return
  • د) print

١٠) علشان تطبع كل عنصر فى المصفوفة واحداً واحداً:

for (let degree ______ degreeList) {
  console.log(degree);
}
  • أ) in
  • ب) of
  • ج) from
  • د) =

القسم الثالث — مفاهيم

١١) عناصر أمن المعلومات الثلاثة هى:

  • أ) السرعة والدقة والتخزين
  • ب) السرية والسلامة والتوافر
  • ج) الأصالة والثبات والانتشار
  • د) الإدخال والمعالجة والإخراج

١٢) عنوان IPv4 بيتكوّن من:

  • أ) ٨ بت
  • ب) ١٦ بت
  • ج) ٣٢ بت
  • د) ١٢٨ بت

١٣) الـ1 KB بيساوى:

  • أ) ١٠٠٠ byte
  • ب) ١٠٢٤ byte
  • ج) ١٠٢٤ bit
  • د) ٨ byte

١٤) التصيّد Phishing معناه:

  • أ) برنامج بينسخ نفسه ويدمّر الملفات
  • ب) رسالة أو موقع مزيّف بينتحل جهة موثوقة علشان ياخد بياناتك
  • ج) تشفير ملفاتك وطلب فدية
  • د) تسجيل ضربات لوحة المفاتيح

١٥) فى قاعدة البيانات العلائقية، الأعمدة فى الجدول اسمها:

  • أ) سجلات Records
  • ب) حقول Fields
  • ج) استعلامات Queries
  • د) علاقات Relations

تفسير النتيجة

دى المصححة الحقيقية، مش الأرقام على الورق.

الدرجةالتفسيرإيه اللى تعمله
٠ – ٤بيبدأ من الصفر تماماً — وده طبيعى ومتوقع، ومش مؤشر ضعف بأى شكل.ده الوضع الافتراضى للكورس أصلاً، والكورس كله متصمّم على أساس الطالب ده. ما تعملش معاه أى حاجة مختلفة، وما تقولّهوش إنه فى فئة.
٥ – ٩فيه بقايا معرفة — بيفتكر لما يشوف، بس مش بينتج.يستفيد أكتر من التمارين المتقدمة فى النص التانى من الكورس. خلّيه يشرح خطوة لزميله من وقت للتانى — بيثبّت عنده وبيرفع ثقته.
١٠ فأكترجاهز للمستوى المتقدم من التمارين من أول محاضرة.ادّيه تمارين المستوى المتقدم كاملة فى كل حصة، ولو خلّص بدرى فى البلوك المرن استخدمه كمساعد. خد بالك: قوى فى التعرّف لا يعنى قوى فى الكتابة — اختبره فى الإنتاج بدرى.
يُقال بصوت عالٍ — قبل التوزيع وبعده

«الورقة دى مقياس بداية مش حكم على حد. أنا محتاج أعرف إحنا واقفين فين علشان أعرف أمشى إزاى. وبصراحة، اللى هيجيب درجة قليلة ده مش غلطته: السنة اللى فاتت المادة كانت نجاح ورسوب وخارج المجموع، فمنطقى إن محدش أخدها بجد. الكورس ده متصمّم أصلاً لطالب مش فاكر حاجة. واللى فاكر — كويس، هيمشى أسرع شوية وبس».

الاحتفاظ بخط الأساس

احتفظ بالأوراق دى. فى المحاضرة ٨ هتدّى نفس نمط الأسئلة تانى، والفرق بين الورقتين هو أقوى حاجة تعرضها لولى الأمر آخر الكورس، وأقوى حاجة يشوفها الطالب عن نفسه.

سجّل عندك رقمين بس لكل طالب: درجة القسم الأول (التتبع) ودرجة القسم التانى (الإكمال). القسم التالت معلومة عامة عن خلفية أولى ثانوى، مش عن استعداده للبرمجة.
