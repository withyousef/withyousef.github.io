المقارنة الصارمة

بتقارن القيمة والنوع مع بعض، و== بتتساهل مع اختلاف النوع.

true

ودى نقطة البداية

كل مقارنة بترجّع true أو false — قيمة من نوع boolean.

وقفنا عند سؤال: عرفنا إن البرنامج بيقدر يقول صح أو غلط. طيب إيه اللى يخلّيه يتصرّف على أساسها — ينفّذ سطراً لو الإجابة true ويتخطّاه لو false؟

الـboolean هو الجسر. والنهارده هنعدّى عليه.

مثال ٠١

كل مقارنة بترجّع true أو false

let mark = 72;

console.log(mark > 50);
console.log(mark === 100);
console.log(mark <= 72);
الناتجtrue false true

مفيش حاجة جديدة هنا — ده بالظبط اللى قفلنا بيه المحاضرة اللى فاتت. المقارنة سؤال، والسؤال ليه إجابة، والإجابة قيمة من نوع boolean.

الجديد هو السؤال التالى: القيمة دى بتروح فين؟ لحد دلوقتى إحنا بس بنطبعها. النهارده هنحطّها فى مكان بيغيّر مسار البرنامج كله.

كل شرط هتكتبه طول عمرك هو فى الآخر تعبير بيرجّع true أو false. لو عرفت تحسب ناتج المقارنة، عرفت تقرا أى شرط.

مثال ٠٢ — أول قرار

أول if فى حياتك

let mark = 72;

if (mark >= 50) {
  console.log("Pass");
}

console.log("Done");
الناتجPass Done

البنية ثلاث قطع: كلمة if، ثم قوس عادى جوّاه الشرط، ثم قوس معقوف جوّاه الكود اللى هيشتغل. ومفيش فاصلة منقوطة بعد القوس العادى.

لو غيّرنا mark لـ30، السطر Pass مش هيتطبع خالص — لكن Done هيتطبع زى ما هو، لأنه برّه الأقواس.

مثال ٠٣ — الطريقان

else — وطيب لو الشرط غلط؟

let mark = 38;

if (mark >= 50) {
  console.log("Pass");
} else {
  console.log("Fail");
}
الناتجFail

كلمة else بتتكتب بعد القوس المعقوف اللى قفل، ومالهاش شرط بين قوسين — لأن شرطها هو «أى حاجة تانية».

مع else بقى عندك ضمان: لازم سطر واحد يتنفّذ. من غيرها كان ممكن البرنامج يعدّى من غير ما يعمل أى حاجة.

مثال ٠٤ — أهم فكرة فى الحصة

else if — سلسلة، مش قائمة

let mark = 72;

if (mark >= 85) {
  console.log("A");
} else if (mark >= 70) {
  console.log("B");
} else if (mark >= 50) {
  console.log("C");
} else {
  console.log("F");
}
الناتجB

لاحظ إن 72 >= 50 صحيحة كمان. ومع ذلك C ما اتطبعتش — لأن السلسلة وقفت عند أول تطابق ومكمّلتش.

وعلشان كده الترتيب بيغيّر النتيجة. لو حطّينا mark >= 50 فى الأول، كل طالب ناجح هياخد C والباقى مش هيتفحص. القاعدة: ابدأ بالأضيق وانزل للأوسع.

مثال ٠٥

شرطان مع بعض: && و||

&& بتقول «الاتنين»، و|| بتقول «واحد على الأقل». وده كل الجدول فى سطر.

let mark = 72;
let attend = 90;

if (mark >= 50 && attend >= 75) {
  console.log("Full pass");
}

if (mark >= 90 || attend >= 90) {
  console.log("Honor");
}
الناتجFull pass Honor

الشرط الأول: الدرجة كويسة والحضور كويس، فالاتنين صح والناتج صح. والتانى: الدرجة مش ٩٠، لكن الحضور ٩٠ — وواحد يكفى مع ||.

العلامتان بيتكتبوا مرتين: && مش &، و|| مش |. والعلامة الواحدة ليها معنى تانى خالص مش هنستخدمه.

مثال ٠٦ — جديد تماماً

وطيب لو عايز أسأل «مش يساوى»؟

العلامة دى ما شفتوهاش قبل كده خالص — لا فى المنصة ولا فى الكتاب. وهى أكتر علامة هتحتاجها من غير ما تتوقّع.

let city = "Cairo";

console.log(city != "Giza");
console.log(city != "Cairo");

if (city !== "Giza") {
  console.log("Not Giza");
}
الناتجtrue false Not Giza

علامة التعجّب هنا معناها النفى. فـ!= تُقرأ «مش يساوى»، وبترجّع true لما القيمتان يكونوا مختلفين — بالظبط عكس ==.

وزى ما عندنا == و===، عندنا != و!==. المتساهلة بتقارن القيمة بس، والصارمة بتقارن القيمة والنوع. والقاعدة نفسها: استخدم الصارمة !== دايماً.

من غير != كنت هتضطر تكتب if فاضية وelse فيها الشغل. العلامة دى بتوفّر عليك السخافة دى.

مثال ٠٧ — جديد تماماً كمان

العلامة ! لوحدها: القلّاب

let isOpen = false;

console.log(isOpen);
console.log(!isOpen);

if (!isOpen) {
  console.log("Closed");
}
الناتجfalse true Closed

علامة التعجّب قدام قيمة منطقية بتقلبها: !true بتبقى false، و!false بتبقى true. وبس — مالهاش أى وظيفة تانية.

ليه دى مفيدة؟ لأن أسماء المتغيرات المنطقية بتتكتب بالإيجاب عادةً — isOpen مش isClosed. فلما تحتاج تسأل عن العكس، بتحط ! قدامها بدل ما تعمل متغيراً تانياً.

اقراها بصوت عالٍ كده: !isOpen يعنى «مش مفتوح». وده أقرب للغة البشر من isOpen == false.

مثال ٠٨

شرط جوّه شرط

let mark = 72;
let attend = 60;

if (mark >= 50) {
  if (attend >= 75) {
    console.log("Pass");
  } else {
    console.log("Low attendance");
  }
} else {
  console.log("Fail");
}
الناتجLow attendance

الشرط الخارجى صح، فدخلنا جوّه. وجوّه لقينا شرطاً تانى، وهو غلط، فمشينا فى else بتاعه. الشرط الخارجى else ما اتفتحش أصلاً.

  • كل if جوّه if بيزيد مستوى إزاحة جديداً. الإزاحة مش زينة — هى اللى بتوريك أنت جوه أنهى قوس.
  • القوس المعقوف اللى بيقفل بيقابل آخر واحد فتح. عُدّهم بإصبعك لو اتلخبطت.
  • الكود ده ممكن يتكتب بـ&& فى سطر واحد — لكن ساعتها هتخسر رسالة «الحضور قليل»، لأن التعشيش بيفرّق بين نوعى الرسوب.
تحذير

مستويان كفاية. لو لقيت نفسك بتكتب if جوه if جوه if، وقّف — الكود ده هيبقى صعب عليك أنت شخصياً بعد أسبوع. أعِد ترتيب الشروط أو استخدم &&.

المهارة — مش معلومة

التتبّع اليدوى: تمشى مكان الكمبيوتر سطراً بسطر

let a = 5;
let b = 12;

a = a + 3;

if (a > b) {
  b = b - a;
} else {
  b = b + a;
}

console.log(a);
console.log(b);
الناتج8 20

الجدول جنبه هو الطريقة. عمود لكل متغير، وسطر لكل خطوة، وما تكتبش الناتج قبل ما توصل لآخر سطر.

  • القيمة الجديدة بتتكتب فى سطر جديد، والقديمة ما تتمسحش — علشان تشوف الحكاية كلها.
  • سطر الشرط ما بيغيّرش أى قيمة. بيغيّر السطر اللى بعده بس.
  • لو الشرط غلط، اشطب الفرع اللى ما اتنفّذش بخط واحد وكمّل.

المهارة دى هى الفرق بين طالب بيخمّن الإجابة وطالب بيوصل لها. وهى مطلوبة صراحةً فى الامتحان لما يقولوا «تتبّع الكود التالى».

مثال ٠٩ — الفخ

علامة يساوى واحدة جوّه الشرط

let mark = 40;

if (mark = 50) {
  console.log("Pass");
}

console.log(mark);
الناتجPass 50

الدرجة ٤٠ والشرط بيقول ٥٠، ومع ذلك طبع Pass. والأسوأ: mark نفسها اتغيّرت لـ50 من غير ما حد يطلب.

السبب إن = معناها «حُطّ فى» زى ما اتعلّمنا فى المحاضرة الأولى. فالسطر ده مش بيسأل — هو بيحطّ ٥٠ فى mark، والشرط بياخد الناتج ويعتبره صح.

والصح إنك تسأل بعلامتين أو تلاتة، مش بواحدة:

if (mark === 50) {
  console.log("Pass");
}

وده خطأ منطقى صامت: مفيش رسالة خطأ، والبرنامج بيشتغل لآخره، وبيدّى نتايج غلط بهدوء. اسأل نفسك قبل كل شرط: أنا بسأل ولا بحطّ؟

الشرط مش بيحسب حاجة. هو بيختار أنهى سطر يشتغل.

لحد المحاضرة اللى فاتت كان البرنامج بيمشى فى خط واحد من فوق لتحت، ما بيفوّتش سطراً ولا بيرجع. النهارده بقى ليه خرائط — وأنت اللى بترسمها بالشرط.

  • قرار واحد: if.
  • طريقان: if وelse.
  • سلّم بيقف عند أول تطابق: else if.
  • ربط وقلب: && و|| و! و!==.
  • ومهارة واحدة تحت كل ده: التتبّع اليدوى.

المحاضرة الجاية: الحلقات — إزاى نخلّى البرنامج يكرّر الشغل من غير ما نكتبه مية مرة.

الواجب

أربع تمارين — من ١٥ لـ٢٠ دقيقة

يتسلّم مكتوباً بخط اليد فى أول المحاضرة الجاية. اكتب الحل على ورق، مش على التابلت.

٠١

تتبّع بالورقة والقلم

اعمل جدولاً بعمودين a وb، واكتب قيمة كل واحد بعد كل سطر، وفى الآخر اكتب اللى هيتطبع.

let a = 10;
let b = 4;

if (a > b) {
  a = a - b;
} else {
  b = b - a;
}

console.log(a);
console.log(b);
٠٢

اكتب سلسلة شروط

عندك متغير temp فيه درجة الحرارة. اكتب سلسلة تطبع Hot لو أكبر من ٣٥، وWarm لو من ٢٥ لـ٣٥، وCool لو أقل من ٢٥. وجرّبها بثلاث قيم من عندك.

٠٣

فين الغلط؟

فيه غلطتان. حدّدهما واكتب الصح، والمطلوب إن الكود يطبع Fail للقيمة دى.

let mark = 30;

if (mark = 50) {
  console.log("Pass");
}
  console.log("Fail");
٠٤

صح ولا غلط

اكتب قدام كل تعبير true أو false، وجملة واحدة تشرح ليه.

  • 7 != 7
  • "a" !== "b"
  • !(5 > 3)
  • 10 > 5 && 3 > 8
  • 10 > 5 || 3 > 8
مهمة جانبية — دقيقتان، لا تُسلّم
